package com.nexus.aicontrol;

import android.app.NotificationChannel;
import android.app.NotificationManager;
import android.content.Context;
import android.hardware.camera2.CameraManager;
import android.media.AudioManager;
import android.os.Build;
import android.os.Handler;
import android.os.Looper;
import android.widget.Toast;

import androidx.core.app.NotificationCompat;

import com.getcapacitor.JSObject;
import com.getcapacitor.Plugin;
import com.getcapacitor.PluginCall;
import com.getcapacitor.PluginMethod;
import com.getcapacitor.annotation.CapacitorPlugin;

/**
 * Нативный модуль Nexus AI Control.
 * Фонарик, громкость, системный тост, уведомления.
 * Доступен из JS как Capacitor.Plugins.NexusNative
 */
@CapacitorPlugin(name = "NexusNative")
public class NexusNativePlugin extends Plugin {

    private boolean torchOn = false;

    /** Фонарик: { on: true|false } */
    @PluginMethod
    public void torch(PluginCall call) {
        Boolean on = call.getBoolean("on", Boolean.TRUE);
        boolean want = (on != null) && on;
        try {
            CameraManager cm = (CameraManager) getContext().getSystemService(Context.CAMERA_SERVICE);
            String id = cm.getCameraIdList()[0];
            cm.setTorchMode(id, want);
            torchOn = want;
            JSObject r = new JSObject();
            r.put("on", want);
            call.resolve(r);
        } catch (Exception e) {
            call.reject("torch: " + e.getMessage());
        }
    }

    /** Громкость медиа: { level: 0..100 } */
    @PluginMethod
    public void volume(PluginCall call) {
        Integer level = call.getInt("level", 50);
        int want = (level == null) ? 50 : level;
        try {
            AudioManager am = (AudioManager) getContext().getSystemService(Context.AUDIO_SERVICE);
            int max = am.getStreamMaxVolume(AudioManager.STREAM_MUSIC);
            int v = Math.round(max * Math.max(0, Math.min(100, want)) / 100f);
            am.setStreamVolume(AudioManager.STREAM_MUSIC, v, 0);
            JSObject r = new JSObject();
            r.put("level", Math.round(v * 100f / max));
            call.resolve(r);
        } catch (Exception e) {
            call.reject("volume: " + e.getMessage());
        }
    }

    /** Системный тост: { text: "..." } */
    @PluginMethod
    public void toast(PluginCall call) {
        final String text = call.getString("text", "");
        new Handler(Looper.getMainLooper()).post(new Runnable() {
            @Override
            public void run() {
                Toast.makeText(getContext(), text, Toast.LENGTH_LONG).show();
            }
        });
        call.resolve();
    }

    /** Уведомление: { title, body } */
    @PluginMethod
    public void notify(PluginCall call) {
        String title = call.getString("title", "Nexus");
        String body = call.getString("body", "");
        try {
            NotificationManager nm = (NotificationManager) getContext().getSystemService(Context.NOTIFICATION_SERVICE);
            String ch = "nexus_control";
            if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
                nm.createNotificationChannel(new NotificationChannel(ch, "Nexus Control", NotificationManager.IMPORTANCE_HIGH));
            }
            NotificationCompat.Builder b = new NotificationCompat.Builder(getContext(), ch)
                    .setSmallIcon(android.R.drawable.ic_dialog_info)
                    .setContentTitle(title)
                    .setContentText(body)
                    .setAutoCancel(true);
            nm.notify((int) (System.currentTimeMillis() % 100000), b.build());
            call.resolve();
        } catch (Exception e) {
            call.reject("notify: " + e.getMessage());
        }
    }

    /** Состояние модуля */
    @PluginMethod
    public void info(PluginCall call) {
        JSObject r = new JSObject();
        r.put("torch", torchOn);
        r.put("model", Build.MODEL);
        r.put("sdk", Build.VERSION.SDK_INT);
        call.resolve(r);
    }
}
