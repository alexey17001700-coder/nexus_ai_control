package com.nexus.aicontrol;

import android.app.Notification;
import android.app.NotificationChannel;
import android.app.NotificationManager;
import android.app.Service;
import android.content.Context;
import android.content.Intent;
import android.content.pm.ServiceInfo;
import android.graphics.Bitmap;
import android.graphics.PixelFormat;
import android.hardware.display.DisplayManager;
import android.hardware.display.VirtualDisplay;
import android.media.Image;
import android.media.ImageReader;
import android.media.projection.MediaProjection;
import android.media.projection.MediaProjectionManager;
import android.os.Build;
import android.os.IBinder;
import android.util.DisplayMetrics;
import android.view.WindowManager;

import java.io.ByteArrayOutputStream;
import java.nio.ByteBuffer;

/**
 * Служба захвата экрана через MediaProjection.
 * Держит VirtualDisplay + ImageReader, хранит последний кадр.
 * Обязательна для Android 14+ (foreground service type mediaProjection).
 */
public class ScreenCaptureService extends Service {

    public static ScreenCaptureService instance;

    private final Object lock = new Object();
    private MediaProjection projection;
    private VirtualDisplay vdisplay;
    private ImageReader reader;
    private int width, height, density;
    private volatile Bitmap lastFrame;
    private volatile long lastStamp = 0;

    @Override
    public void onCreate() {
        super.onCreate();
        instance = this;
    }

    @Override
    public int onStartCommand(Intent intent, int flags, int startId) {
        startForegroundNotif();
        if (intent != null) {
            int code = intent.getIntExtra("resultCode", 0);
            Intent data = intent.getParcelableExtra("data");
            if (code != 0 && data != null && projection == null) {
                startCapture(code, data);
            }
        }
        return START_STICKY;
    }

    private void startForegroundNotif() {
        String ch = "nexus_screen";
        NotificationManager nm = (NotificationManager) getSystemService(Context.NOTIFICATION_SERVICE);
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
            nm.createNotificationChannel(new NotificationChannel(ch, "Nexus Screen", NotificationManager.IMPORTANCE_LOW));
        }
        Notification n = new Notification.Builder(this, ch)
                .setContentTitle("Nexus: трансляция экрана")
                .setContentText("Экран транслируется")
                .setSmallIcon(android.R.drawable.ic_menu_view)
                .build();
        if (Build.VERSION.SDK_INT >= 29) {
            startForeground(9911, n, ServiceInfo.FOREGROUND_SERVICE_TYPE_MEDIA_PROJECTION);
        } else {
            startForeground(9911, n);
        }
    }

    private void startCapture(int resultCode, Intent data) {
        try {
            MediaProjectionManager mpm = (MediaProjectionManager) getSystemService(Context.MEDIA_PROJECTION_SERVICE);
            projection = mpm.getMediaProjection(resultCode, data);

            WindowManager wm = (WindowManager) getSystemService(Context.WINDOW_SERVICE);
            DisplayMetrics dm = new DisplayMetrics();
            wm.getDefaultDisplay().getRealMetrics(dm);
            width = dm.widthPixels;
            height = dm.heightPixels;
            density = dm.densityDpi;

            reader = ImageReader.newInstance(width, height, PixelFormat.RGBA_8888, 2);
            vdisplay = projection.createVirtualDisplay("nexus_screen",
                    width, height, density,
                    DisplayManager.VIRTUAL_DISPLAY_FLAG_AUTO_MIRROR,
                    reader.getSurface(), null, null);

            reader.setOnImageAvailableListener(new ImageReader.OnImageAvailableListener() {
                @Override
                public void onImageAvailable(ImageReader r) {
                    Image img = null;
                    try {
                        img = r.acquireLatestImage();
                        if (img == null) return;
                        Image.Plane[] planes = img.getPlanes();
                        ByteBuffer buf = planes[0].getBuffer();
                        int pixelStride = planes[0].getPixelStride();
                        int rowStride = planes[0].getRowStride();
                        int rowPadding = rowStride - pixelStride * width;
                        int bw = width + rowPadding / pixelStride;
                        Bitmap bmp = Bitmap.createBitmap(bw, height, Bitmap.Config.ARGB_8888);
                        bmp.copyPixelsFromBuffer(buf);
                        Bitmap cropped = Bitmap.createBitmap(bmp, 0, 0, width, height);
                        bmp.recycle();
                        synchronized (lock) {
                            Bitmap old = lastFrame;
                            lastFrame = cropped;
                            lastStamp = System.currentTimeMillis();
                            if (old != null && !old.isRecycled()) old.recycle();
                        }
                    } catch (Exception e) {
                        // кадр пропущен
                    } finally {
                        if (img != null) img.close();
                    }
                }
            }, null);
        } catch (Exception e) {
            // не удалось запустить захват
        }
    }

    /** JPEG последнего кадра (синхронно, потокобезопасно). */
    public byte[] captureJpeg(int quality) {
        synchronized (lock) {
            if (lastFrame == null || lastFrame.isRecycled()) return null;
            ByteArrayOutputStream bos = new ByteArrayOutputStream();
            lastFrame.compress(Bitmap.CompressFormat.JPEG, quality, bos);
            return bos.toByteArray();
        }
    }

    public int getWidth() { return width; }
    public int getHeight() { return height; }
    public long getStamp() { return lastStamp; }

    public void stopCapture() {
        try { if (vdisplay != null) vdisplay.release(); } catch (Exception e) {}
        try { if (reader != null) reader.close(); } catch (Exception e) {}
        try { if (projection != null) projection.stop(); } catch (Exception e) {}
        vdisplay = null; reader = null; projection = null;
        synchronized (lock) {
            Bitmap f = lastFrame; lastFrame = null; lastStamp = 0;
            if (f != null && !f.isRecycled()) f.recycle();
        }
    }

    @Override
    public void onDestroy() {
        stopCapture();
        instance = null;
        super.onDestroy();
    }

    @Override
    public IBinder onBind(Intent intent) { return null; }
}
